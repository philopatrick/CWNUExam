\section{Part A: Multiple Choice Questions (15 \times 3 = 45)}

\begin{question}
A random experiment is one which:
\paren
\begin{choices}
\item Has a fixed outcome
\item Has more than one possible outcome
\item Always gives equal outcomes
\item Has probability zero
\end{choices}
\end{question}

\begin{question}
The sample space of tossing two coins contains:
\paren
\begin{choices}
\item 2
\item 3
\item 4
\item 6
\end{choices}
\end{question}

\begin{question}
If events $A$ and $B$ are independent, then:
\paren
\begin{choices}
\item $P(A \cup B)=1$
\item $P(A\mid B)=P(A)$
\item $P(A)=P(B)$
\item $P(A\cap B)=0$
\end{choices}
\end{question}

\begin{question}
The number of permutations of 5 objects taken 2 at a time is:
\paren
\begin{choices}
\item 10
\item 20
\item 25
\item 120
\end{choices}
\end{question}

\begin{question}
The value of $\binom{6}{2}$ is:
\paren
\begin{choices}
\item 6
\item 12
\item 15
\item 30
\end{choices}
\end{question}

\begin{question}
A Bernoulli random variable can take:
\paren
\begin{choices}
\item Any real values
\item Only positive integers
\item Values 0 and 1
\item Values 1 and 2
\end{choices}
\end{question}

\begin{question}
If $X\sim\mathrm{Bin}(n,p)$, then $E[X]$ equals:
\paren
\begin{choices}
\item $p$
\item $np$
\item $np(1-p)$
\item $1/p$
\end{choices}
\end{question}

\begin{question}
A Poisson distribution is used when:
\paren
\begin{choices}
\item $n$ is small, $p$ is large
\item $n$ is large, $p$ is small
\item $n$ and $p$ are both large
\item $n=1$
\end{choices}
\end{question}

\begin{question}
For a continuous random variable $X$:
\paren
\begin{choices}
\item $P(X=a)>0$
\item $P(X=a)=1$
\item $P(X=a)=0$
\item $P(X=a)$ is undefined
\end{choices}
\end{question}

\begin{question}
The uniform distribution on $(0,1)$ has density:
\paren
\begin{choices}
\item $f(x)=x$
\item $f(x)=e^{-x}$
\item $f(x)=1$
\item $f(x)=2x$
\end{choices}
\end{question}

\begin{question}
Variance can be written as:
\paren
\begin{choices}
\item $E[X]$
\item $E[X^2]-(E[X])^2$
\item $E[X+1]$
\item $(E[X])^2$
\end{choices}
\end{question}

\begin{question}
If $X$ and $Y$ are independent, $\mathrm{Cov}(X,Y)$ equals:
\paren
\begin{choices}
\item 1
\item $-1$
\item 0
\item Depends on $X,Y$
\end{choices}
\end{question}

\begin{question}
The Central Limit Theorem applies when:
\paren
\begin{choices}
\item Sample size is small
\item Sample size tends to infinity
\item Variance is zero
\item Distribution is normal
\end{choices}
\end{question}

\begin{question}
Maximum Likelihood Estimation chooses parameters that:
\paren
\begin{choices}
\item Minimize error
\item Maximize likelihood
\item Minimize variance
\item Maximize sample size
\end{choices}
\end{question}

\begin{question}
Least squares estimation minimizes:
\paren
\begin{choices}
\item Absolute errors
\item Squared errors
\item Probability
\item Variance
\end{choices}
\end{question}

\section{Part A: Multiple Choice Questions (15 \times 3 = 45)}

\begin{question}
An event in probability is:
\paren
\begin{choices}
\item A number
\item A random variable
\item A subset of the sample space
\item A probability
\end{choices}
\end{question}

\begin{question}
The sample space for rolling a die once has:
\paren
\begin{choices}
\item 1 outcome
\item 6 outcomes
\item 12 outcomes
\item Infinite outcomes
\end{choices}
\end{question}

\begin{question}
If $P(A)=0.4$ and $P(B)=0.5$ for independent events, then $P(A\cap B)=$
\paren
\begin{choices}
\item 0.2
\item 0.9
\item 0.4
\item 0.5
\end{choices}
\end{question}

\begin{question}
The number of combinations of 5 objects taken 3 at a time is:
\paren
\begin{choices}
\item 10
\item 20
\item 30
\item 60
\end{choices}
\end{question}

\begin{question}
A random variable taking values $0,1,2,\ldots$ is:
\paren
\begin{choices}
\item Bernoulli
\item Binomial
\item Poisson
\item Continuous
\end{choices}
\end{question}

\begin{question}
The mean of a Poisson($\lambda$) random variable is:
\paren
\begin{choices}
\item $1/\lambda$
\item $\lambda$
\item $\lambda^2$
\item $\sqrt{\lambda}$
\end{choices}
\end{question}

\begin{question}
For a Binomial random variable, the variance is:
\paren
\begin{choices}
\item $np$
\item $p(1-p)$
\item $np(1-p)$
\item $n(1-p)$
\end{choices}
\end{question}

\begin{question}
The probability for a continuous random variable is calculated using:
\paren
\begin{choices}
\item Summation
\item Counting
\item Integration
\item Multiplication
\end{choices}
\end{question}

\begin{question}
The exponential distribution is defined on:
\paren
\begin{choices}
\item $(-\infty,\infty)$
\item $(0,1)$
\item $(0,\infty)$
\item $(-1,1)$
\end{choices}
\end{question}

\begin{question}
If $f(x)\ge0$ and $\int f(x),dx=1$, then $f(x)$ is a:
\paren
\begin{choices}
\item Distribution
\item Density function
\item Random variable
\item Probability
\end{choices}
\end{question}

\begin{question}
Covariance measures:
\paren
\begin{choices}
\item Mean
\item Spread
\item Linear relationship
\item Probability
\end{choices}
\end{question}

\begin{question}
If $X$ and $Y$ are independent, then:
\paren
\begin{choices}
\item $E[XY]=E[X]E[Y]$
\item $E[XY]=0$
\item $\mathrm{Var}(XY)=0$
\item $X=Y$
\end{choices}
\end{question}

\begin{question}
The sample mean $\bar X$ is an estimator of:
\paren
\begin{choices}
\item Variance
\item Median
\item Population mean
\item Mode
\end{choices}
\end{question}

\begin{question}
The Central Limit Theorem requires:
\paren
\begin{choices}
\item Normal population
\item Large sample size
\item Discrete variables only
\item Zero variance
\end{choices}
\end{question}

\begin{question}
Least squares estimation is mainly used in:
\paren
\begin{choices}
\item Hypothesis testing
\item Parameter estimation
\item Sampling
\item Simulation
\end{choices}
\end{question}

\end{problem}
